% LaTeX preamble for PDF article
% Author: Frank H Jung

% Page geometry (override pandoc default)
\usepackage{geometry}
\geometry{a4paper, margin=2.5cm}

% Typography
\usepackage{mathpazo}
\usepackage{microtype}
\usepackage{setspace}
\setstretch{1.4}

% Colors and links
\usepackage{xcolor}
\definecolor{accent}{HTML}{1771ad}
\usepackage{hyperref}
\hypersetup{
    colorlinks=true,
    linkcolor=accent,
    urlcolor=accent,
    citecolor=accent
}

% Headers and footers
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0.4pt}
\fancyhead[L]{\nouppercase{\leftmark}}
\fancyhead[R]{\thepage}
\fancyfoot[L]{Frank\textit{ly speaking...}}
\fancyfoot[R]{\textcopyright\ Frank H Jung \the\year}
% Pandoc maps ## headings to \subsection, so update \leftmark
% from \subsectionmark (not just \sectionmark).
\renewcommand{\sectionmark}[1]{\markboth{#1}{}}
\renewcommand{\subsectionmark}[1]{\markboth{#1}{}}

\fancypagestyle{plain}{
  \fancyhf{}
  \renewcommand{\headrulewidth}{0pt}
  \renewcommand{\footrulewidth}{0.4pt}
  \fancyfoot[L]{Frank\textit{ly speaking...}}
  \fancyfoot[R]{\textcopyright\ Frank H Jung \the\year}
  \fancyfoot[C]{\thepage}
}

% Lists
\usepackage{enumitem}
\setlist{topsep=0.5em, itemsep=0.25em}

% Floats and images
\usepackage{float}
% Pandoc defines \maxheight as \textheight, but this does not
% account for figure captions and float spacing. Reduce it to
% leave room for captions and avoid "Float too large" warnings.
\makeatletter
\def\maxheight{\ifdim\Gin@nat@height>0.85\textheight 0.85\textheight\else\Gin@nat@height\fi}
\makeatother

% Captions
\usepackage{caption}
\captionsetup{font=small, labelfont=bf}

% Prevent widows and orphans
\widowpenalty=10000
\clubpenalty=10000
